     6→     6→
     7→     7→type Theme = 'dark' | 'light'
     8→     8→
     9→     9→const ThemeCtx = createContext<{ theme: Theme; toggle: () => void }>({ theme: 'light', toggle: () => {} })
    10→    10→
    11→    11→function useTheme() {
    12→    12→  return useContext(ThemeCtx)
    13→    13→}
    14→    14→
    15→    15→function ThemeProvider({ children }: { children: React.ReactNode }) {
    16→    16→  const [theme, setTheme] = useState<Theme>(() => {
    17→    17→    if (typeof window === 'undefined') return 'light'
    18→    18→    return (localStorage.getItem('ui-theme') as Theme) || 'light'
    19→    19→  })
    20→    20→
    21→    21→  useEffect(() => {
    22→    22→    localStorage.setItem('ui-theme', theme)
    23→    23→    document.documentElement.style.colorScheme = theme
    24→    24→  }, [theme])
    25→    25→
    26→    26→  const toggle = () => setTheme(t => (t === 'dark' ? 'light' : 'dark'))
    27→    27→
    28→    28→  return <ThemeCtx.Provider value={{ theme, toggle }}>{children}</ThemeCtx.Provider>
    29→    29→}
