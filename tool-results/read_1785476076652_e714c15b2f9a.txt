     5→/* ───────────────────────── THEME ───────────────────────── */
     6→
     7→type Theme = 'dark' | 'light'
     8→
     9→const ThemeCtx = createContext<{ theme: Theme; toggle: () => void }>({ theme: 'light', toggle: () => {} })
    10→
    11→function useTheme() {
    12→  return useContext(ThemeCtx)
    13→}
    14→
    15→function ThemeProvider({ children }: { children: React.ReactNode }) {
    16→  const [theme, setTheme] = useState<Theme>(() => {
    17→    if (typeof window === 'undefined') return 'light'
    18→    return (localStorage.getItem('ui-theme') as Theme) || 'light'
    19→  })
    20→
    21→  useEffect(() => {
    22→    localStorage.setItem('ui-theme', theme)
    23→    document.documentElement.style.colorScheme = theme
    24→  }, [theme])
    25→
    26→  const toggle = () => setTheme(t => (t === 'dark' ? 'light' : 'dark'))
    27→
    28→  return <ThemeCtx.Provider value={{ theme, toggle }}>{children}</ThemeCtx.Provider>
    29→}
    30→
    31→function v(theme: Theme) {
    32→  const d = theme === 'dark'
    33→  return {
    34→    bg:       d ? '#000000' : '#FFFFFF',
    35→    text:     d ? '#FFFFFF' : '#000000',
    36→    muted:    d ? 'rgba(255,255,255,0.55)' : 'rgba(0,0,0,0.45)',
    37→    faint:    d ? 'rgba(255,255,255,0.3)' : 'rgba(0,0,0,0.25)',
    38→    ghost:    d ? 'rgba(255,255,255,0.15)' : 'rgba(0,0,0,0.12)',
    39→    divider:  d ? 'rgba(255,255,255,0.1)' : 'rgba(0,0,0,0.08)',
    40→    navBg:    d ? 'rgba(0,0,0,0.9)' : 'rgba(255,255,255,0.9)',
    41→    forest:   d ? 'rgba(255,255,255,0.13)' : 'rgba(0,0,0,0.10)',
    42→  }
    43→}
