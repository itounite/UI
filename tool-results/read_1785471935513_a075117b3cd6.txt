     7→const RED = '#7F1D1D'

    10→type Theme = 'dark' | 'light'
    11→
    12→const ThemeCtx = createContext<{ theme: Theme; toggle: () => void }>({ theme: 'dark', toggle: () => {} })
    13→
    14→function useTheme() {
    15→  return useContext(ThemeCtx)
    16→}
    17→
    18→function ThemeProvider({ children }: { children: React.ReactNode }) {
    19→  const [theme, setTheme] = useState<Theme>(() => {
    20→    if (typeof window === 'undefined') return 'dark'
    21→    const stored = localStorage.getItem('ui-theme') as Theme | null
    22→    return stored || 'dark'
    23→  })
    24→
    25→  useEffect(() => {
    26→    localStorage.setItem('ui-theme', theme)
    27→    document.documentElement.style.colorScheme = theme
    28→  }, [theme])
    29→
    30→  const toggle = () => setTheme(t => (t === 'dark' ? 'light' : 'dark'))
    31→
    32→  return <ThemeCtx.Provider value={{ theme, toggle }}>{children}</ThemeCtx.Provider>
    33→}
